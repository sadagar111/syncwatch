# Libraries SyncWatch needs (the source-code version installs them with SyncWatch.bat).
# The full PySide6 package is needed: PySide6-Essentials alone has no video support.
# Tested with PySide6 6.12; 6.5 is the oldest with the FFmpeg video backend.
PySide6>=6.5
# Checks the signatures on updates (and, from step 3, encrypts rooms).
cryptography>=42
# The video player (from 0.4.2): python-mpv talks to libmpv, which SyncWatch.bat fetches
# into libmpv\ on Windows (tools/get_libmpv.py). Linux: apt install libmpv2.
mpv>=1.0.7
# The built-in converter (from 0.4.2): PyAV, with its own FFmpeg.
av>=14
